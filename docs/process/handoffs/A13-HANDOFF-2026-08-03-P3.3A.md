Each record contains:

```json
{
  "id": "stableUniqueId",
  "status": "ACTIVE",
  "owner": "logical owner",
  "sourceFile": "repository-relative path",
  "entrypoint": "installer or hook entrypoint",
  "process": "system_server|system_ui|launcher|android|per_app|resource|other",
  "phase": "SYSTEM_SERVER_STARTING|PACKAGE_READY|RESOURCE_INIT|BOOTSTRAP|OTHER",
  "preferenceKeys": [],
  "reasonCode": "controlled enum",
  "reason": "specific technical explanation",
  "coveredCallSites": [],
  "hookTargets": [],
  "testEvidence": [],
  "exitCondition": "specific condition for migration or removal"
}
```
